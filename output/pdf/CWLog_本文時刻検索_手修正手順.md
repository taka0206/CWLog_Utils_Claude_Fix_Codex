#### 反映の順序

1　06_確認と復旧で版・バックアップ・文字コードを確認する。

2　上表のS01からS07を順番に修正する。S08は説明コメントの更新なので任意。

3　05_設定例を参照し、運用中の設定ファイルへ抽出形式とタイムゾーンを指定する。

4　06_確認と復旧で構文確認・付属テスト・実ログの照合を行い、反映状態と照合状態を記入する。

#### 検索結果に関わる変更

--start または --end 指定時に本文日時で判定します。--hoursだけの場合は既存のイベント時刻による取得です。

期間指定時は全期間の候補をFilterLogEventsで取得します。ログ量に応じて時間・通信・一時領域が増えます。

--engine insightsも期間指定時はfilterに切り替わり、logs:FilterLogEvents権限が必要です。

--limitは本文判定の後に適用します。Excelの日時列と並び順はイベント時刻です。

日時を読み取れないイベントは除外し、件数を警告します。開始・終了の両端を含み、ミリ秒単位で比較します。

# --- 本文時刻による期間検索 (--start / --end 指定時) --------------------------
cw_fetch_via_message_time() {
    local group=$1 start_ms=$2 end_ms=$3 pattern=$4 stream=$5 raw=$6
    local regex=${CFG_events_message_time_regex:-} regex_rc=0
    [[ "" =~ $regex ]]
    regex_rc=$?
    if [[ -z $regex ]] || ((regex_rc == 2)); then
        cw_say_err "[エラー] events_message_time_regex に有効な正規表現を指定してください (第1括弧で日時全体を取得)。"
        return 1
    fi
    local message_tz=${CFG_events_message_timezone:-${CFG_events_timezone:-}}
    if [[ -n $message_tz ]]; then
        if [[ -d /usr/share/zoneinfo && ! -f /usr/share/zoneinfo/$message_tz ]]; then
            cw_say_err "[エラー] 本文日時のタイムゾーンが見つかりません: $message_tz"
            return 1
        fi
        local TZ=$message_tz
        export TZ
    fi

    local all="$CW_TMPDIR/events_all.tsv"
    : > "$all" || return 1
    cw_filter_range "$group" 0 0 "$pattern" 0 "$stream" "$all" || return $?

    local row message value parsed seconds millis stamp skipped=0
    while IFS= read -r row; do
        message=${row#*$'\t'}
        message=${message#*$'\t'}
        # @tsv のエスケープを戻してから、最初に一致する日時を抽出する。
        printf -v message '%b' "$message"
        if [[ ! $message =~ $regex ]] || [[ -z ${BASH_REMATCH[1]:-} ]]; then
            skipped=$((skipped + 1))
            continue
        fi
        value=${BASH_REMATCH[1]//\//-}
        value=${value/,/.}
        if ! parsed=$(date -d "$value" '+%s %3N' 2>/dev/null); then
            skipped=$((skipped + 1))
            continue
        fi
        read -r seconds millis <<< "$parsed"
        stamp=$((seconds * 1000 + 10#$millis))
        if ((stamp >= start_ms && stamp <= end_ms)); then
            printf '%s\n' "$row" || return 1
        fi
    done < "$all" > "$raw" || return 1
    if ((skipped > 0)); then
        cw_say_warn "[警告] 本文日時を抽出・解析できない ${skipped} 件を除外しました。events_message_time_regex / events_message_timezone を確認してください。"
    fi
    return 0
}
